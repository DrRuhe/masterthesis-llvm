    ROUND(MAX(cv_spec), 1) as max_cv_spec

    SELECT kernel,
           STDDEV(t_spec_ns)/AVG(t_spec_ns)*100 as cv_spec
    FROM v_ablation_results
    WHERE study_name='corpus_uc_final_20260610'
      AND config_name='default'
      AND t_spec_ns IS NOT NULL
    GROUP BY kernel
    HAVING COUNT(*) >= 3
) x;
